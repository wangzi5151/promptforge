  -H "Content-Type: application/json" \
  -H "X-API-Key: pf_xxx" \
  -d '{"title":"翻译助手","body":"将以下 {{lang}} 文本翻译成中文：{{text}}"}'

curl -X POST http://localhost:8000/api/prompts/{id}/run \
  -H "X-API-Key: pf_xxx" \
  -d '{"variables":{"lang":"English","text":"hello"},"call_llm":true}'
